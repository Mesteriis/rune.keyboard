cmake_minimum_required(VERSION 3.31.6)
project(rune_scoring_evaluation LANGUAGES C CXX)

set(RUNE_LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../runtime-llama/src/main/cpp/llama.cpp")
execute_process(COMMAND git -C "${RUNE_LLAMA_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE RUNE_LLAMA_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT RUNE_LLAMA_COMMIT STREQUAL "36b10154383b60eb15baac2c7a40d2a5f784faa7")
    message(FATAL_ERROR "The evaluation requires the pinned llama.cpp revision")
endif()
execute_process(COMMAND git -C "${RUNE_LLAMA_SOURCE}" status --porcelain --untracked-files=all
    OUTPUT_VARIABLE RUNE_LLAMA_CHANGES OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT RUNE_LLAMA_CHANGES STREQUAL "")
    message(FATAL_ERROR "The reference evaluation requires clean pinned llama.cpp sources")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES
    LLAMA_BUILD_SERVER LLAMA_BUILD_APP GGML_METAL GGML_CUDA GGML_VULKAN GGML_OPENMP
    GGML_NATIVE GGML_BACKEND_DL GGML_CPU_KLEIDIAI GGML_LLAMAFILE GGML_CCACHE)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${RUNE_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_library(rune_scoring STATIC scoring.cpp)
target_compile_features(rune_scoring PUBLIC cxx_std_17)
target_compile_options(rune_scoring PRIVATE -Wall -Wextra -Werror)
target_link_libraries(rune_scoring PUBLIC llama)
target_include_directories(rune_scoring PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
add_executable(rune-score score_main.cpp)
target_compile_options(rune-score PRIVATE -Wall -Wextra -Werror)
target_include_directories(rune-score SYSTEM PRIVATE "${RUNE_LLAMA_SOURCE}/vendor")
target_link_libraries(rune-score PRIVATE rune_scoring)
enable_testing()
add_executable(scoring-math-test scoring_math_test.cpp)
target_compile_options(scoring-math-test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(scoring-math-test PRIVATE rune_scoring)
add_test(NAME scoring-math COMMAND scoring-math-test)
add_executable(scoring-model-test scoring_model_test.cpp)
target_compile_options(scoring-model-test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(scoring-model-test PRIVATE rune_scoring)
